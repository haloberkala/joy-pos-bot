                </div>

              )}
              
              <div className="border-t pt-4">

                </div>

                    className="gap-2"
                    onClick={() => handleDeletePurchase(viewPurchase)}
                  >
                    <Trash2 className="w-4 h-4" /> Hapus Riwayat

                )}
                <Button variant="outline" onClick={() => setViewPurchase(null)}>
                  Tutup
