                      <div key={item.id} className="flex justify-between text-sm py-2 border-b border-slate-200 dark:border-slate-800 last:border-0">

                      variant="destructive" 
                      size="sm" 
                      className="gap-2"
                      onClick={() => handleDeletePurchase(viewPurchase)}
                    >
                      <Trash2 className="w-4 h-4" /> Hapus Riwayat
                    </Button>
                  )}
                </div>
                <Button variant="outline" onClick={() => { setViewPurchase(null); setIsEditingNote(false); }}>
                  Tutup
                </Button>
              </div>
            </>
